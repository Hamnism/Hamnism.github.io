{{- /* 記事末尾の前後ナビ。同じ区分の読む順で前と次を出す。 */ -}} {{- $list := partial "ordered.html" .Section -}} {{- $prev := false }}{{ $next := false }}{{ $n := len $list -}} {{- range $i, $p := $list }}{{ if eq $p $ }} {{- if gt $i 0 }}{{ $prev = index $list (sub $i 1) }}{{ end }} {{- if lt (add $i 1) $n }}{{ $next = index $list (add $i 1) }}{{ end }} {{- end }}{{ end -}} {{- if or $prev $next }}
{{ with $prev }}前の記事{{ .Title }}{{ else }}{{ end }} {{ with $next }}次の記事{{ .Title }}{{ else }}{{ end }}
{{- end }}
